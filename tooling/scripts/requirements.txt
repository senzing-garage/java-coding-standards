# Runtime dependencies for the canonical Java formatter
# (`tooling/scripts/format_java.py`).
#
# Install locally with:
#     pip install -r tooling/scripts/requirements.txt
#
# Python 3.10+ is required (tree-sitter declares
# `Requires-Python >=3.10` as of the 0.25 line). The pinned
# versions below are kept tight so that parses are deterministic
# across developer machines and CI; bumps go through the
# standards-repo dependabot cooldown and the calibration-gate
# re-run.
#
# IMPORTANT — every pin here is mirrored by the `GRAMMAR_VERSION`
# dict in `format_java.py`, and `TestGrammarVersionPins` fails the
# build when the two drift. Dependabot can only edit this file, so
# its bump PRs always arrive red; bump `GRAMMAR_VERSION` in the
# same commit to make them green.

# tree-sitter Python binding. 0.26.0 was released 2026-06-30 and
# was still the latest on PyPI when checked on 2026-10-16. The
# release date is a fact; "latest" is not, so it carries the date
# it was checked rather than claiming to be current forever.
# Dependabot will say when that changes. Pins like this also
# constrain Dependabot — set the range carefully when bumping.
tree-sitter==0.26.0

# tree-sitter-java grammar. 0.23.5 was released 2024-12-21 and was
# still the latest on PyPI when checked on 2026-10-16. The
# formatter dispatches on grammar node names; the emitter assumes
# the specific node-name set this version emits, so bumping
# requires a calibration-gate run.
tree-sitter-java==0.23.5
